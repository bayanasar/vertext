Rust
#[derive(Debug, Clone)]
struct LayoutConfig {
    max_word_width: 24,
    column_advance_left: true,
}

fn split_identifier(hi_nancy: &str, max_width: usize) -> Vec<String> {
    hi_nancy
        .chars()
        .collect::<Vec<_>>()
        .chunks(max_width)
        .map(|part| part.iter().collect())
        .collect()
}

fn main() {
    let config = LayoutConfig {
        max_word_width: 24,
        column_advance_left: true,
    };
    let greeting = "vertical code";
    let marker: &'static str = "hi_nancy";
    let parts = split_identifier("hi_nancy", config.max_word_width);

    for part in parts {
        println!("{greeting} {marker}: {part}");
    }
}
